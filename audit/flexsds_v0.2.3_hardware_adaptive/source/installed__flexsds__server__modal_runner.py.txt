110:                 if match and not endpoint:

112:             output.capture(proc, on_line=observe_endpoint, close_on_exit=not deploy)
113:             proc._flexsds_output = output

115:             def ready():

118:                 while not endpoint:

130:                                             config=cfg, deploy=deploy)
131:                 output.notice(f"Inference verified: {endpoint[0]}/v1 (model={model})")
132:                 output.update('Serving ready' if not deploy else 'Deployment verified',100)
133:                 if deploy:
134:                     output.close()
135:             proc._flexsds_readiness = ready
