19: class CostProfile:
20:     """Hardware cost profile lookup table C(k, B) in milliseconds."""
21:     matrix: Dict[Tuple[int, int], float] = field(default_factory=dict)

23:     def get_cost(self, k: int, batch_size: int = 1) -> float:
24:         """Get step latency in ms for draft depth k and batch size.
25:         
26:         Falls back to matching k if exact batch is missing, or a linear heuristic.
27:         """
28:         if (k, batch_size) in self.matrix:
29:             return self.matrix[(k, batch_size)]
30: 
31:         # Fallback 1: Tìm key có cùng k
32:         matching_k = [cost for (cand_k, _), cost in self.matrix.items() if cand_k == k]
33:         if matching_k:
34:             return matching_k[0]
35: 
36:         # Fallback 2: Ước lượng tuyến tính mặc định nếu chưa có profile
37:         base_t = 20.0  # ms for vanilla step
38:         draft_step_t = 3.0  # ms for draft step
39:         return base_t if k == 0 else (base_t + k * draft_step_t)

42: class EmpiricalCostProfile(CostProfile):
43:     """Use the closest measured batch for an observed K; never invent costs."""
44:     def get_cost(self, k: int, batch_size: int = 1) -> float:
45:         batches = [b for cand_k, b in self.matrix if cand_k == k]
46:         if not batches:
47:             raise ValueError(f'No measured cost for K={k}')
48:         closest = min(batches, key=lambda b: (abs(b-batch_size), b))
49:         return self.matrix[(k, closest)]

56:         self,

59:         initial_alpha: float = 0.8,

61:         high_threshold: float = 0.80,

67:         self.window_size = window_size
68:         self.ema_beta = ema_beta
69:         self.alpha_ema = initial_alpha
70:         self.fallback_threshold = fallback_threshold
71:         self.high_threshold = high_threshold
72:         self.candidate_budgets = candidate_budgets or [0, 2, 3, 4]
73:         self.cost_profile = cost_profile or CostProfile()
74:         self.min_free_vram_mb = min_free_vram_mb
75: 
76:         self.initial_alpha = initial_alpha
77:         self._window: deque[float] = deque(maxlen=window_size)
78:         if history_limit is not None and (type(history_limit) is not int or history_limit < 1):
79:             raise ValueError("history_limit must be a positive integer")
80:         self._history = [] if history_limit is None else deque(maxlen=history_limit)

82:     def get_rolling_alpha(self) -> float:
83:         """Returns the current Exponential Moving Average of alpha."""
84:         return self.alpha_ema

86:     def estimate_expected_tokens(self, k: int, alpha: float) -> float:
87:         """Calculates expected token generation per verification step: E[tau] = 1 + sum_(i=1..k) (alpha^i)."""
88:         if k <= 0:
89:             return 1.0
90:         gain = 1.0
91:         curr_power = 1.0
92:         for _ in range(k):
93:             curr_power *= alpha
94:             gain += curr_power
95:         return gain
96: 
97:     def update(self, num_accepted: int, num_drafted: int) -> float:
98:         """Updates rolling alpha telemetry from the latest step verification."""
99:         if num_drafted <= 0:
100:             step_alpha = self.alpha_ema
101:         else:
102:             step_alpha = min(max(num_accepted / num_drafted, 0.0), 1.0)
103: 
104:         self._window.append(step_alpha)
105:         self.alpha_ema = self.ema_beta * self.alpha_ema + (1.0 - self.ema_beta) * step_alpha
106:         return self.alpha_ema

108:     def select_budget(
109:         self,
110:         batch_size: int = 1,
111:         free_vram_mb: float = 16000.0,
112:         kv_cache_usage_pct: float = 0.0,
113:     ) -> BudgetDecision:
114:         """Selects optimal budget k* in candidate_budgets based on VRAM, alpha, and cost model.
115: 
116:         kv_cache_usage_pct is accepted for interface parity with AdaptiveDraftController
117:         and is not used by the v0 base controller.
118:         """
119:         alpha = self.alpha_ema
120: 
121:         # 1. Memory Pressure Rule (Chống OOM)
122:         if free_vram_mb < self.min_free_vram_mb:
123:             decision = BudgetDecision(
124:                 k=0,
125:                 alpha_ema=alpha,
126:                 expected_tokens=1.0,
127:                 reason=f"VRAM pressure: free {free_vram_mb:.1f}MB < min {self.min_free_vram_mb:.1f}MB",
128:             )
129:             self._history.append({"k": 0, "alpha": alpha, "reason": decision.reason})
130:             return decision
131: 
132:         # 2. Safety Fallback Gate (Triệt tiêu tăng tốc âm khi alpha thấp)
133:         if alpha < self.fallback_threshold:
134:             decision = BudgetDecision(
135:                 k=0,
136:                 alpha_ema=alpha,
137:                 expected_tokens=1.0,
138:                 reason=f"Low acceptance fallback: alpha {alpha:.3f} < threshold {self.fallback_threshold:.3f}",
139:             )
140:             self._history.append({"k": 0, "alpha": alpha, "reason": decision.reason})
141:             return decision
142: 
143:         # 3. Cost-Model Argmax kết hợp Heuristic Guards
144:         c_vanilla = self.cost_profile.get_cost(k=0, batch_size=batch_size)
145:         vanilla_efficiency = 1.0 / max(c_vanilla, 1e-4)
146: 
147:         best_k = 0
148:         best_efficiency = vanilla_efficiency
149:         best_expected_tokens = 1.0
150: 
151:         for cand_k in self.candidate_budgets:
152:             if cand_k == 0:
153:                 continue
154: 
155:             # Heuristic guard: Không mở rộng k=4 trừ khi alpha >= high_threshold (0.80)
156:             if cand_k == 4 and alpha < self.high_threshold:
157:                 continue
158: 
159:             e_tau = self.estimate_expected_tokens(cand_k, alpha)
160:             c_cand = self.cost_profile.get_cost(k=cand_k, batch_size=batch_size)
161:             efficiency = e_tau / max(c_cand, 1e-4)
162: 
163:             # Heuristic mid-alpha: nếu alpha < high_threshold (0.80) và cand_k == 2,
164:             # ưu tiên k=2 hơn k=3 nếu hiệu năng ngang ngửa để giảm tải tính toán
165:             if efficiency > best_efficiency:
166:                 best_efficiency = efficiency
167:                 best_k = cand_k
168:                 best_expected_tokens = e_tau
169: 
170:         reason = (
171:             "Vanilla fallback (efficiency bound)"
172:             if best_k == 0
173:             else f"Optimal budget k={best_k} with eff={best_efficiency:.4f}"
174:         )
175: 
176:         decision = BudgetDecision(
177:             k=best_k,
178:             alpha_ema=alpha,
179:             expected_tokens=best_expected_tokens,
180:             reason=reason,
181:         )
182:         self._history.append({"k": best_k, "alpha": alpha, "reason": decision.reason})
183:         return decision
184: 
185:     def reset(self, initial_alpha: Optional[float] = None) -> None:
186:         """Resets controller state when beginning a new sequence/prompt."""
187:         if initial_alpha is not None:
188:             self.alpha_ema = initial_alpha
189:             self.initial_alpha = initial_alpha
190:         self._window.clear()
191:         self._history.clear()
192: 
193:     def get_stats(self) -> dict:
194:         """Exports runtime telemetry matching Interface A contract."""
195:         tree_budgets = [h["k"] for h in self._history]
196:         return {
197:             "alpha_ema": self.alpha_ema,
198:             "tree_budgets": tree_budgets,
199:             "step_count": len(self._window),
200:             "mean_k": (sum(tree_budgets) / len(tree_budgets)) if tree_budgets else 0.0,
201:             "history": list(self._history),
202:         }

205: class AdaptiveDraftController(BudgetController):
206:     """Runtime Adaptive Draft Controller for vLLM speculative decoding.
207: 
208:     Extends BudgetController with:
209:     - 10-step rolling window tracking raw acceptance samples
210:     - Unbiased EMA warm-up correction for fast reaction to low acceptance rate
211:     - Modular budget rules (co giãn k in {0, 2, 3, 4})
212:     - Memory pressure rule (KV-cache usage + VRAM guard) with cooldown and hysteresis
213:     - Self-diagnostic window statistics (mean, variance, min, max, is_warmed_up)
214:     """

217:         self,
218:         window_size: int = 10,
219:         ema_beta: float = 0.8,
220:         initial_alpha: float = 0.8,

222:         high_threshold: float = 0.80,
223:         candidate_budgets: Optional[List[int]] = None,
224:         cost_profile: Optional[CostProfile] = None,
225:         min_free_vram_mb: float = 1500.0,
226:         probe_interval: int = 20,
227:         probe_k: int = 2,

229:         kv_cache_recover_pct: float = 80.0,

260:         self.enable_vram_guard = enable_vram_guard

263:         self._memory_pressure_active = False
264:         self._memory_pressure_steps = 0

272:         """Returns the list of raw alpha samples in the current sliding window."""

275:     def get_effective_alpha(self) -> float:
276:         """Returns unbiased alpha taking into account warm-up correction.
277: 
278:         If step_count == 0, returns initial_alpha.
279:         When step_count > 0:
280:         unbiased_alpha = (alpha_ema - initial_alpha * (ema_beta ** step_count)) / (1.0 - (ema_beta ** step_count))
281:         Clamped to [0.0, 1.0].
282:         """
283:         if self.step_count == 0:
284:             return self.initial_alpha
285: 
286:         beta_power = self.ema_beta ** self.step_count
287:         denominator = 1.0 - beta_power
288:         if denominator <= 1e-6:
289:             return self.alpha_ema
290: 
291:         unbiased = (self.alpha_ema - self.initial_alpha * beta_power) / denominator
292:         return min(max(unbiased, 0.0), 1.0)
293: 
294:     def get_window_stats(self) -> dict:
295:         """Returns self-diagnostic statistics of the rolling window."""
296:         samples = list(self._window)
297:         is_warmed_up = self.step_count >= self.window_size
298: 
299:         if not samples:
300:             return {
301:                 "mean": self.initial_alpha,
302:                 "min": self.initial_alpha,
303:                 "max": self.initial_alpha,
304:                 "variance": 0.0,
305:                 "is_warmed_up": False,
306:                 "samples": [],
307:                 "step_count": self.step_count,
308:             }
309: 
310:         mean_val = sum(samples) / len(samples)
311:         var_val = sum((x - mean_val) ** 2 for x in samples) / len(samples)
312: 
313:         return {
314:             "mean": mean_val,
315:             "min": min(samples),
316:             "max": max(samples),
317:             "variance": var_val,
318:             "is_warmed_up": is_warmed_up,
319:             "samples": samples,
320:             "step_count": self.step_count,
321:         }

323:     def apply_memory_pressure_rule(
324:         self,
325:         free_vram_mb: float,
326:         kv_cache_usage_pct: float = 0.0,
327:     ) -> Optional[Tuple[int, str]]:
328:         """Memory pressure rule with hysteresis, evaluated once per decode step.
329: 
330:         - Trigger: KV-cache usage > kv_cache_high_pct, or (VRAM guard enabled)
331:           free VRAM < min_free_vram_mb -> force k=0 and (re)start cooldown.
332:         - Hold: k=0 for at least memory_cooldown_steps steps (trigger step included).
333:         - Release: after cooldown, only when KV-cache usage < kv_cache_recover_pct
334:           and free VRAM is back above the minimum; Budget Rule then takes over.
335: 
336:         kv_cache_usage_pct is in percent (0-100), e.g. 100 * scheduler.get_kv_cache_usage().
337:         Returns (k, reason) while memory pressure holds k=0, else None.
338:         """
339:         vram_low = self.enable_vram_guard and free_vram_mb < self.min_free_vram_mb
340:         kv_high = kv_cache_usage_pct > self.kv_cache_high_pct
341: 
342:         if kv_high or vram_low:
343:             self._memory_pressure_active = True
344:             self._memory_pressure_steps = 1
345:             if kv_high:
346:                 reason = (
347:                     f"KV-cache pressure: usage {kv_cache_usage_pct:.1f}% > "
348:                     f"high {self.kv_cache_high_pct:.1f}%"
349:                 )
350:             else:
351:                 reason = (
352:                     f"VRAM pressure: free {free_vram_mb:.1f}MB < "
353:                     f"min {self.min_free_vram_mb:.1f}MB"
354:                 )
355:             return (0, reason)
356: 
357:         if not self._memory_pressure_active:
358:             return None
359: 
360:         cooldown_done = self._memory_pressure_steps >= self.memory_cooldown_steps
361:         if cooldown_done and kv_cache_usage_pct < self.kv_cache_recover_pct:
362:             self._memory_pressure_active = False
363:             self._memory_pressure_steps = 0
364:             return None
365: 
366:         self._memory_pressure_steps += 1
367:         if not cooldown_done:
368:             reason = (
369:                 f"Memory pressure cooldown: step {self._memory_pressure_steps}/"
370:                 f"{self.memory_cooldown_steps}"
371:             )
372:         else:
373:             reason = (
374:                 f"Memory pressure hold: KV-cache {kv_cache_usage_pct:.1f}% >= "
375:                 f"recover {self.kv_cache_recover_pct:.1f}%"
376:             )
377:         return (0, reason)

379:     def apply_budget_rule(self, alpha: float) -> Tuple[int, str]:
380:         """Core budget rule adapting k in {0, 2, 3, 4} based on empirical alpha thresholds.
381: 
382:         - alpha >= high_threshold (0.80): maintain k=3 (or expand to k=4 if alpha >= 0.90)
383:         - fallback_threshold <= alpha < high_threshold: throttle down to k=2
384:         - alpha < fallback_threshold (0.65): fallback to k=0 with periodic probing (k=probe_k every probe_interval steps)
385:         """
386:         if alpha < self.fallback_threshold:
387:             self.consecutive_fallback_steps += 1
388:             if self.probe_interval > 0 and self.consecutive_fallback_steps >= self.probe_interval:
389:                 # Kích hoạt probing để thoát bẫy kẹt Vanilla vĩnh viễn
390:                 self.consecutive_fallback_steps = 0
391:                 return (
392:                     self.probe_k,
393:                     f"Periodic probe: reached {self.probe_interval} fallback steps (alpha={alpha:.3f}), probing with k={self.probe_k}",
394:                 )
395:             hold_str = f" (hold {self.consecutive_fallback_steps}/{self.probe_interval})" if self.probe_interval > 0 else ""
396:             return (
397:                 0,
398:                 f"Low acceptance fallback: alpha {alpha:.3f} < threshold {self.fallback_threshold:.3f}{hold_str}",
399:             )
400: 
401:         # alpha >= self.fallback_threshold -> reset bộ đếm fallback
402:         self.consecutive_fallback_steps = 0
403: 
404:         if alpha >= self.high_threshold:
405:             if alpha >= 0.90 and 4 in self.candidate_budgets:
406:                 return (4, f"High acceptance (aggressive): alpha {alpha:.3f} >= 0.90, selected k=4")
407:             return (3, f"High acceptance: alpha {alpha:.3f} >= threshold {self.high_threshold:.3f}, maintain k=3")
408: 
409:         # fallback_threshold <= alpha < high_threshold
410:         return (
411:             2,
412:             f"Mid acceptance: alpha {alpha:.3f} in [{self.fallback_threshold:.2f}, {self.high_threshold:.2f}), throttled to k=2",
413:         )

415:     def select_budget(
416:         self,
417:         batch_size: int = 1,
418:         free_vram_mb: float = 16000.0,
419:         kv_cache_usage_pct: float = 0.0,
420:     ) -> BudgetDecision:
421:         """Selects optimal budget k* using unbiased alpha and modular rules."""
422:         effective_alpha = self.get_effective_alpha()
423: 
424:         # 1. Memory Pressure Rule (Chống OOM)
425:         mem_decision = self.apply_memory_pressure_rule(
426:             free_vram_mb=free_vram_mb,
427:             kv_cache_usage_pct=kv_cache_usage_pct,
428:         )
429:         if mem_decision is not None:
430:             k, reason = mem_decision
431:             decision = BudgetDecision(
432:                 k=k,
433:                 alpha_ema=effective_alpha,
434:                 expected_tokens=1.0,
435:                 reason=reason,
436:             )
437:             self._history.append({"k": k, "alpha": effective_alpha, "reason": reason})
438:             return decision
439: 
440:         # 2. Budget Rule (Co giãn k theo alpha kết hợp Periodic Probing)
441:         k, reason = self.apply_budget_rule(alpha=effective_alpha)
442:         e_tau = self.estimate_expected_tokens(k, effective_alpha)
443: 
444:         decision = BudgetDecision(
445:             k=k,
446:             alpha_ema=effective_alpha,
447:             expected_tokens=e_tau,
448:             reason=reason,
449:         )
450:         self._history.append({"k": k, "alpha": effective_alpha, "reason": reason})
451:         return decision
