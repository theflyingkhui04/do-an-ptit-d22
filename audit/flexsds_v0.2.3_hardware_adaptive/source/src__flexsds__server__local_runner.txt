18: def finish_launch(proc, background: bool = False) -> int:
19:     """Propagate startup/foreground failure without equating a process with readiness."""
20:     if proc is None:
21:         print("Launch failed; no serving readiness established.")
22:         return 1
23:     output = getattr(proc, '__dict__', {}).get('_flexsds_output')
24:     if output is None:
25:         print("Serving process started; verifying endpoint readiness...")
26:     readiness = getattr(proc, '__dict__', {}).get('_flexsds_readiness')
27:     try:
28:         if readiness is not None:
29:             readiness()
30:         else:
31:             print("Endpoint readiness is not verified for this process.")
32:         session = getattr(proc, '__dict__', {}).get('_flexsds_session')
33:         if session and output:
34:             from flexsds.cli.serving_session import present_session
35:             present_session(proc, output, **session)
36:         code = proc.poll() if background else proc.wait()
37:         return code if isinstance(code, int) else 0
38:     except (RuntimeError, OSError, ValueError) as exc:
39:         if output:
40:             output.update('Serving failed')
41:             output.close()
42:             print(f"Full details: {output.log_path}")
43:         print(f"Serving readiness failed: {exc}")
44:         proc.terminate()
45:         try:
46:             proc.wait(timeout=15)
47:         except subprocess.TimeoutExpired:
48:             proc.kill()

233:         effective_port = port if port is not None else config.get("serving", config).get("port", 8000)
234:         if not dry_run and not self.check_port_available(effective_port):
235:             print(f"[Error] Port {effective_port} is already in use; choose another port or run flexsds stop.")
236:             return None

238:         env = self.setup_environment(config)
239: 
240:         if dry_run:
241:             print("\n[FlexSDS] Dry-run execution command:")
242:             print(" ".join(cmd))
243:             print(f"Environment: ADAPTIVE_ENABLE_VRAM_GUARD={env.get('ADAPTIVE_ENABLE_VRAM_GUARD')}")
244:             return None
245: 
246:         if config.get('serving', config).get('enable_adaptive', True):
247:             from flexsds.core.hardware import detect_hardware, matches_gpu_label
248:             actual = detect_hardware()
249:             data = json.loads(Path(config.get('serving', config)['profile_path']).read_text())
250:             if not actual.is_cuda_available or (actual.sm_architecture != data['metadata']['sm_architecture']
251:                                                or not matches_gpu_label(data['metadata']['device'], actual.device_name)):
252:                 raise ValueError('Profile differs from actual serving GPU')
253: 
254:         if background and os.name == "nt":

256:             return None

272:                 from flexsds.server.readiness import verify_endpoint
273:                 def ready():

277:                 return proc
278:             else:
279:                 proc = subprocess.Popen(cmd, env=env)
280:                 from flexsds.server.readiness import verify_endpoint
281:                 def ready():
282:                     model = verify_endpoint(f"http://127.0.0.1:{effective_port}", process=proc)

290:     def stop(self, pid_file: str = ".flexsds.pid") -> bool:
291:         """Dừng nhóm tiến trình máy chủ vLLM và dọn dẹp sạch tiến trình con."""
292:         if os.name == "nt":
293:             print("Local process-group stop is unsupported on Windows; stop the foreground session.")
294:             return False
295:         if not os.path.exists(pid_file):
296:             return False
297: 
298:         try:
299:             with open(pid_file, "r", encoding="utf-8") as f:
300:                 content = f.read().strip()
301:                 if not content:
302:                     return False
303:                 pid = int(content)
304: 
305:             os.killpg(pid, signal.SIGTERM)
306:         except (ProcessLookupError, ValueError):
307:             pass
308:         finally:
309:             if os.path.exists(pid_file):
310:                 os.remove(pid_file)
311: 
312:         return True
