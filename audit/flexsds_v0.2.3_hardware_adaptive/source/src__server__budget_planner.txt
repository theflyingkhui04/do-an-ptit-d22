1: """vLLM-free budget planning logic behind the Lane 3 adaptive scheduler.
2: 
3: The vLLM scheduler subclass (``src.server.adaptive_scheduler``) only forwards
4: runtime signals to this planner, so every decision rule here can be unit-tested
5: without a GPU or a vLLM install.
6: """
7: from __future__ import annotations
8: 
9: import json
10: import logging
11: import math
12: import os
13: import time
14: from copy import deepcopy
15: from dataclasses import dataclass, field
16: from typing import Dict, Iterable, Mapping, Optional, Set, Tuple
17: 
18: from src.proposer.adapter import AdaptiveProposerAdapter
19: from src.proposer.controller import BudgetDecision

21: # Free VRAM value passed to the controller when no reading is available,
22: # so the VRAM guard never fires on a missing signal.
23: NO_VRAM_READING_MB = 1.0e9
24: logger = logging.getLogger(__name__)

27: @dataclass
28: class StepPlan:
29:     """Planned nodes and scalar budgets; neither proves tree execution.
30: 
31:     Existing batch_k/per_request_k retain their capped scalar semantics.
32:     Decisions are detached snapshots of the project-owned controller types.
33:     """
34:     batch_k: int
35:     per_request_k: Dict[str, int] = field(default_factory=dict)
36:     batch_node_count: int = 0
37:     per_request_node_count: Dict[str, int] = field(default_factory=dict)
38:     per_request_decisions: Dict[str, BudgetDecision] = field(default_factory=dict)
39:     plan_sequence: Optional[int] = None
40: 
41: 
42: class AdaptiveBudgetPlanner:
43:     """Turns per-request AdaptiveDraftController decisions into a batch plan."""
44: 
45:     def __init__(
46:         self,
47:         adapter: Optional[AdaptiveProposerAdapter] = None,
48:         *,
49:         max_k: int = 4,
50:         min_batch_k: int = 0,
51:         per_request: bool = True,
52:         telemetry_path: Optional[str] = None,
53:         step_telemetry_path: Optional[str] = None,
54:     ):
55:         if max_k < 0 or min_batch_k < 0:
56:             raise ValueError("max_k and min_batch_k must be non-negative")
57:         if (telemetry_path and step_telemetry_path
58:                 and os.path.normcase(os.path.realpath(telemetry_path))
59:                 == os.path.normcase(os.path.realpath(step_telemetry_path))):
60:             raise ValueError("Step and aggregate telemetry must use separate paths")
61:         self.adapter = adapter or AdaptiveProposerAdapter()
62:         self.max_k = max_k
63:         self.min_batch_k = min_batch_k
64:         self.per_request = per_request
65:         self.telemetry_path = telemetry_path
66:         self.step_telemetry_path = step_telemetry_path
67:         self._known_requests: Set[str] = set()
68:         self._last_decisions: Dict[str, dict] = {}
69:         self._scheduled_runtime_modes: Dict[str, str] = {}
70:         self._plan_sequence = 0
71: 
72:     def plan_step(
73:         self,
74:         decode_req_ids: Iterable[str],
75:         batch_size: int,
76:         kv_usage: float,
77:         free_vram_mb: Optional[float] = None,
78:     ) -> StepPlan:
79:         """Selects k_i for every decoding request and the batch-wide K = max(k_i).
80: 
81:         kv_usage is the vLLM fraction in [0, 1] (Scheduler.get_kv_cache_usage()).
82:         """
83:         kv_pct = 100.0 * kv_usage
84:         vram = NO_VRAM_READING_MB if free_vram_mb is None else free_vram_mb
85: 
86:         per_request_k: Dict[str, int] = {}
87:         nodes: Dict[str, int] = {}
88:         decisions: Dict[str, BudgetDecision] = {}
89:         alpha_signals: Dict[str, Tuple[Optional[float], Optional[float]]] = {}
90:         for req_id in decode_req_ids:
91:             self._known_requests.add(req_id)
92:             controller = self.adapter.get_controller(req_id)
93:             decision = controller.select_budget(
94:                 batch_size=batch_size,
95:                 free_vram_mb=vram,
96:                 kv_cache_usage_pct=kv_pct,
97:             )
98:             decisions[req_id] = deepcopy(decision)
99:             effective_alpha = getattr(controller, "get_effective_alpha", None)
100:             alpha_signals[req_id] = (getattr(controller, "alpha_ema", None),
101:                                      effective_alpha() if callable(effective_alpha) else None)
102:             nodes[req_id] = max(0, getattr(decision, "num_nodes", decision.k))
103:             per_request_k[req_id] = min(nodes[req_id], self.max_k)
104: 
105:         if not per_request_k:
106:             return StepPlan(batch_k=0, per_request_k={})
107: 
108:         batch_k = max(max(per_request_k.values()), self.min_batch_k)
109:         batch_k = min(batch_k, self.max_k)
110:         if not self.per_request:
111:             per_request_k = {req_id: batch_k for req_id in per_request_k}
112:         batch_node_count = max(nodes.values())
113:         self._plan_sequence += 1
114:         for req_id, decision in decisions.items():
115:             raw_alpha, effective_alpha = alpha_signals[req_id]
116:             record = _decision_record(decision, nodes[req_id], batch_size, kv_pct, free_vram_mb,
117:                                       raw_alpha=raw_alpha, effective_alpha=effective_alpha)
118:             record.update(request_id=req_id, lane="adaptive", record_type="planned_decision",
119:                           controller_class=f"{type(self.adapter.get_controller(req_id)).__module__}.{type(self.adapter.get_controller(req_id)).__name__}",
120:                           plan_sequence=self._plan_sequence, scalar_runtime_budget=per_request_k[req_id],
121:                           batch_scalar_budget=batch_k, batch_node_count=batch_node_count)
122:             self._last_decisions[req_id] = record
123:             self._write_step(record)
124:         return StepPlan(batch_k=batch_k, per_request_k=per_request_k,
125:                         batch_node_count=batch_node_count, per_request_node_count=nodes,
126:                         per_request_decisions=decisions, plan_sequence=self._plan_sequence)
127: 
128:     def record_scheduled_budget(self, req_id: str, batch_k: int, placeholder_k: int,
129:                                 *, dynamic: bool, schedule_sequence: int,
130:                                 plan_sequence: Optional[int]) -> None:
131:         """Observe this output's proposer K and placeholders for future verification.
132: 
133:         K is consumed by the worker handling THIS output. Its proposed tokens
134:         are verified by a later output; neither observation proves GPU execution.
135:         """
136:         self._known_requests.add(req_id)
137:         mode = "linear_dynamic_k" if dynamic else "linear_fixed_k"
138:         self._scheduled_runtime_modes[req_id] = mode
139:         self._write_step({"record_type": "scheduled_budget", "request_id": req_id,
140:                           "lane": "adaptive", "executed_runtime_mode": "unverified",
141:                           "scheduled_runtime_mode": mode,
142:                           "scheduled_scalar_k": batch_k,
143:                           "schedule_sequence": schedule_sequence, "plan_sequence": plan_sequence,
144:                           "budget_target": "drafts_for_future_verification",
145:                           # Compatibility aliases: 'next' refers to verification,
146:                           # not to the worker invocation that consumes this K.
147:                           "next_batch_scalar_budget": batch_k,
148:                           "next_request_placeholder_budget": placeholder_k,
149:                           "executed_draft_token_count": None, "tree_execution_verified": False})
150: 
151:     def _write_step(self, record: dict) -> None:
152:         if self.step_telemetry_path:
153:             try:
154:                 line = json.dumps(dict(record, recorded_at=time.time()), allow_nan=False)
155:                 with open(self.step_telemetry_path, "a", encoding="utf-8") as fh:
156:                     fh.write(line + "\n")
157:             except (OSError, ValueError):
158:                 logger.exception("AdaptiveBudgetPlanner: step telemetry write failed")
159: 
160:     def record_lineage_failure(self, req_id: str, reason: str) -> None:
161:         logger.warning("Adaptive feedback skipped for %s: %s", req_id, reason)
162:         self._write_step({"record_type": "verification_feedback_skipped", "request_id": req_id,
163:                           "lane": "adaptive", "lineage_status": reason,
164:                           "source_plan_sequence": None, "source_schedule_sequence": None,
165:                           "tree_execution_verified": False})
166: 
167:     def record_verification(self, req_id: str, num_drafted: int, num_accepted: int,
168:                             *, source=None) -> None:
169:         """Feeds one verification result back to the request's controller.
170: 
171:         Direct planner callers supply synchronous counts. Production scheduler
172:         callers MUST validate execution provenance before invoking this method.
173:         A newly completed prefill may have an exact schedule but no decode plan.
174:         """
175:         if num_drafted <= 0:
176:             return
177:         self._known_requests.add(req_id)
178:         alpha_before = (self.adapter.get_telemetry(req_id)["alpha_ema"]
179:                         if self.step_telemetry_path else None)
180:         self.adapter.on_verification_completed(
181:             request_id=req_id,
182:             num_accepted=num_accepted,
183:             num_drafted=num_drafted,
184:         )
185:         if self.step_telemetry_path:
186:             self._write_step({
187:                 "record_type": "verification_feedback", "request_id": req_id,
188:                 "lane": "adaptive", "executed_draft_token_count": num_drafted,
189:                 "accepted_draft_token_count": num_accepted,
190:                 "executed_runtime_mode": "unverified",
191:                 "source_plan_sequence": source.plan_sequence if source else None,
192:                 "source_schedule_sequence": source.schedule_sequence if source else None,
193:                 "source_request_scalar_k": source.request_k if source else None,
194:                 "source_batch_scalar_k": source.batch_k if source else None,
195:                 "lineage_status": "attributed" if source else "unverified",
196:                 "tree_execution_verified": False,
197:                 "alpha_ema": self.adapter.get_telemetry(req_id)["alpha_ema"],
198:                 "alpha_ema_before": alpha_before,
199:             })
200: 
201:     def finish_request(self, req_id: str) -> Optional[dict]:
202:         """Releases the request session and appends its telemetry as one JSONL line."""
203:         if req_id not in self._known_requests:
204:             return None
205:         self._known_requests.discard(req_id)
206:         last_decision = self._last_decisions.pop(req_id, {})
207:         mode = self._scheduled_runtime_modes.pop(req_id, "unverified")
208:         telemetry = self.adapter.end_request(req_id)
209:         # Top-level v0.2 decision fields describe the LAST plan only. Legacy
210:         # alpha/history/mean_accept_len remain request aggregates.
211:         telemetry.update({key: value for key, value in last_decision.items()
212:                           if key not in telemetry and key != "record_type"})
213:         telemetry["last_planned_decision"] = deepcopy(last_decision) or None
214:         telemetry.update(record_type="request_aggregate", decision_scope="last_planned_decision",
215:                          scheduled_runtime_mode=mode, executed_runtime_mode="unverified",
216:                          executed_draft_token_count=None,
217:                          tree_execution_verified=False)
218:         if self.telemetry_path:
219:             record = dict(telemetry)
220:             record["finished_at"] = time.time()
221:             with open(self.telemetry_path, "a", encoding="utf-8") as fh:
222:                 fh.write(json.dumps(record) + "\n")
223:         return telemetry

225:     @classmethod
226:     def from_env(cls, environ: Optional[Mapping[str, str]] = None) -> "AdaptiveBudgetPlanner":
227:         """Builds a planner from ADAPTIVE_* environment variables (read once at server start)."""
228:         env = os.environ if environ is None else environ
229: 
230:         raw_kwargs = env.get("ADAPTIVE_CONTROLLER_KWARGS", "").strip()
231:         try:
232:             controller_kwargs = json.loads(raw_kwargs) if raw_kwargs else {}
233:         except json.JSONDecodeError as exc:
234:             raise ValueError(f"ADAPTIVE_CONTROLLER_KWARGS is not valid JSON: {exc}") from exc
235:         if not isinstance(controller_kwargs, dict):
236:             raise ValueError("ADAPTIVE_CONTROLLER_KWARGS must be a JSON object")
237: 
238:         profile_path = env.get("FLEXSDS_PROFILE_PATH")
239:         if profile_path:
240:             import hashlib
241:             from pathlib import Path
242:             from flexsds.core.cost_model import validate_measured_profile
243:             from src.proposer.controller import EmpiricalCostProfile
244:             raw = Path(profile_path).read_bytes()
245:             data = validate_measured_profile(json.loads(raw))
246:             matrix = {(int(key.split('_')[0][1:]), int(key.split('_')[1][1:])): value
247:                       for key, value in data['cost_matrix_ms'].items()}
248:             controller_kwargs['cost_profile'] = EmpiricalCostProfile(matrix=matrix)
249:             controller_kwargs['candidate_budgets'] = data['grid']['candidate_budgets']
250:             logger.info("Loaded empirical hardware profile %s sha256=%s", profile_path,
251:                         hashlib.sha256(raw).hexdigest())
252: 
253:         if "ADAPTIVE_ENABLE_VRAM_GUARD" in env:
254:             controller_kwargs["enable_vram_guard"] = _env_flag(env["ADAPTIVE_ENABLE_VRAM_GUARD"])
255: 
256:         return cls(
257:             adapter=AdaptiveProposerAdapter(controller_kwargs=controller_kwargs),
258:             max_k=int(env.get("ADAPTIVE_MAX_K", "4")),
259:             min_batch_k=int(env.get("ADAPTIVE_MIN_BATCH_K", "0")),
260:             per_request=_env_flag(env.get("ADAPTIVE_PER_REQUEST", "1")),
261:             telemetry_path=env.get("ADAPTIVE_TELEMETRY_PATH") or None,
262:             step_telemetry_path=env.get("ADAPTIVE_STEP_TELEMETRY_PATH") or None,
263:         )
264: 
265: 
266: def _env_flag(value: str) -> bool:
267:     return value.strip().lower() in {"1", "true", "yes", "on"}

270: def _decision_record(decision: BudgetDecision, nodes: int, batch_size: int,
271:                      kv_pct: float, free_vram_mb: Optional[float], *,
272:                      raw_alpha: Optional[float], effective_alpha: Optional[float]) -> dict:
273:     """Additive Interface A v0.2 decision fields (spec §4.6).
274: 
275:     Never derive a topology from a scalar count. Unknown identities keep their
276:     name/count and null structure. Canonical linear serving has no topology;
277:     this module must not load the dormant future-tree catalog for telemetry.
278:     Unmeasured latency, hardware identity and tau are explicitly unavailable.
279:     """
280:     def finite(value: Optional[float]) -> Optional[float]:
281:         return value if value is not None and math.isfinite(value) else None
282: 
283:     name = getattr(decision, "topology_name", None)
284:     topology = None
285:     if name is not None:
286:         topology = {"name": name, "total_nodes_N": nodes,
287:                     "depth_d": None, "parent_indices": None}
288:     status = getattr(decision, "status", None)
289:     return {
290:         "engine_version": None, "hardware_profile": None,
291:         "concurrency_batch_b": batch_size, "acceptance_rate_ema": finite(raw_alpha),
292:         "effective_alpha": finite(effective_alpha),
293:         "dynamic_alpha_star": finite(getattr(decision, "alpha_star", None)),
294:         "selected_topology": topology, "planned_draft_node_count": nodes,
295:         "mean_accepted_tokens_tau": None,
296:         "theoretical_speedup": finite(getattr(decision, "speedup", None)),
297:         "actual_step_latency_ms": None,
298:         "memory_status": {"free_vram_mb": finite(free_vram_mb),
299:                           "kv_cache_usage_pct": finite(kv_pct),
300:                           "guard_triggered": status == "memory_guard" if status else None},
301:         "decision_reason": decision.reason, "executed_runtime_mode": "unverified",
302:         "executed_draft_token_count": None, "tree_execution_verified": False,
303:     }
