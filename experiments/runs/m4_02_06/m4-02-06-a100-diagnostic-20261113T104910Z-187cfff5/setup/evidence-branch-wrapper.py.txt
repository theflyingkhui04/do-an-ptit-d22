import os, subprocess, sys
from pathlib import Path

EVIDENCE_BRANCH = os.environ["M4_EVIDENCE_BRANCH"]

import experiments.m4_git_evidence as ge
ge.BRANCH = EVIDENCE_BRANCH

import experiments.run_m4_colab_campaign as rc

def verify_diagnostic_persisted(directory, evidence_repo):
    relative = Path("experiments/runs/m4_02_06") / directory.name
    for path in directory.rglob("*"):
        if path.is_file():
            target = (relative / path.relative_to(directory)).as_posix()
            data = subprocess.check_output(["git", "show", f"HEAD:{target}"], cwd=evidence_repo)
            if rc.sha256(data) != rc.sha256(path.read_bytes()):
                raise ValueError("Diagnostic not preserved exactly in evidence checkout")
    with ge.git_auth() as env:
        subprocess.run(["git", "fetch", "origin", EVIDENCE_BRANCH],
                       cwd=evidence_repo, env=env, check=True)
    subprocess.run(["git", "merge-base", "--is-ancestor", "HEAD",
                    f"origin/{EVIDENCE_BRANCH}"], cwd=evidence_repo, check=True)

rc.verify_diagnostic_persisted = verify_diagnostic_persisted
rc.publish = ge.publish
rc.main()
